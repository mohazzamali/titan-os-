use axum::{routing::{get, post}, Router, response::Html};
use tokio::net::TcpStream;
use tokio::io::{AsyncWriteExt, AsyncReadExt};
use std::fs;

#[tokio::main]
async fn main() {
    let app = Router::new()
        .route("/", get(serve_dashboard))
        .route("/api/deploy", post(deploy_wasm_payload));

    println!("=======================================================");
    println!("☁️ NEXUS CLOUD WEB SERVER RUNNING ON http://localhost:3000");
    println!("=======================================================");

    let listener = tokio::net::TcpListener::bind("0.0.0.0:3000").await.unwrap();
    axum::serve(listener, app).await.unwrap();
}

async fn serve_dashboard() -> Html<&'static str> {
    Html(include_str!("index.html"))
}

async fn deploy_wasm_payload() -> (axum::http::StatusCode, String) {
    let wasm_path = "../nexus-ai-contract/target/wasm32-unknown-unknown/release/nexus_ai_contract.wasm";
    
    let wasm_payload = match fs::read(wasm_path) {
        Ok(data) => data,
        Err(_) => return (axum::http::StatusCode::NOT_FOUND, "[ERROR] AI WASM FILE NOT FOUND. COMPILE IT FIRST.".to_string()),
    };

    let mut frame = Vec::new();
    frame.extend_from_slice(&[0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF]); 
    frame.extend_from_slice(&[0x99, 0x88, 0x77, 0x66, 0x55, 0x44]); 
    frame.push(0x88); frame.push(0xB5); 
    frame.push(0x03); 
    frame.extend_from_slice(&wasm_payload); 

    match TcpStream::connect("127.0.0.1:9999").await {
        Ok(mut stream) => {
            if stream.write_all(&frame).await.is_ok() {
                // 🚀 TWO-WAY SYNC: Read execution result back from OS Bare-Metal Node!
                let mut response_buf = [0u8; 64];
                match stream.read(&mut response_buf).await {
                    Ok(n) if n > 0 => {
                        let result_msg = String::from_utf8_lossy(&response_buf[..n]);
                        (axum::http::StatusCode::OK, format!("[OK] EXECUTED ON SWARM! RESULT: {}", result_msg))
                    }
                    _ => (axum::http::StatusCode::OK, format!("[OK] DEPLOYED {} BYTES TO AI ENGINE (ASYNC EXECUTION)", wasm_payload.len()))
                }
            } else {
                (axum::http::StatusCode::INTERNAL_SERVER_ERROR, "[ERROR] FAILED TO TRANSMIT TO HARDWARE.".to_string())
            }
        }
        Err(_) => {
            (axum::http::StatusCode::SERVICE_UNAVAILABLE, "[ERROR] NODE OFFLINE. BOOT QEMU OS FIRST.".to_string())
        }
    }
}

use std::fs::{self, File};
use std::io::Write;
use std::path::Path;
use ed25519-compact::{KeyPair, Seed};
use rand::RngCore;

fn main() {
    println!("==================================================");
    println!("⚡ NEXUS TITAN: PROVIDER NODE INSTALLER (MINER)");
    println!("==================================================");

    let efi_nexus_dir = Path::new("/tmp/nexus_efi_install"); 
    fs::create_dir_all(efi_nexus_dir).unwrap();

    println!("[1/3] Copying Bare-Metal Kernel to USB Sector...");
    let kernel_path = Path::new("../nexus-core/target/x86_64-unknown-none/release/bootimage-nexus-core.bin");
    if kernel_path.exists() {
        fs::copy(kernel_path, efi_nexus_dir.join("nexus-kernel.efi")).unwrap();
        println!("  -> Kernel flashed successfully.");
    } else {
        println!("  -> WARNING: Kernel binary not pre-compiled (run cargo bootimage in nexus-core later).");
    }

    println!("[2/3] Generating Unique Node Identity (Wallet)...");
    let mut random_bytes = [0u8; 32];
    rand::thread_rng().fill_bytes(&mut random_bytes);
    
    let keypair = KeyPair::from_seed(Seed::new(random_bytes));
    let public_key = keypair.pk.as_ref();
    
    println!("  -> Node Swarm ID Generated: {:X?}", &public_key[0..8]);
    
    let wallet_path = efi_nexus_dir.join("WALLET.DAT");
    let mut wallet_file = File::create(&wallet_path).unwrap();
    wallet_file.write_all(&random_bytes).unwrap();
    println!("  -> WALLET.DAT securely written to physical drive.");

    println!("[3/3] Finalizing Bootloader Configurations...");
    println!("==================================================");
    println!("🎉 MINER USB READY! Plug this into any PC, boot it, and start earning NXC.");
}

    let mut data_port: Port<u8> = Port::new(0x40);

    let divisor = (PIT_DIVIDEND / target_hz) as u16;

        // Command byte: Channel 0, LSB/MSB access, Mode 2 (Rate Generator), Binary mode
        command_port.write(0x36);
        
        // Send Low Byte of divisor
        data_port.write((divisor & 0xFF) as u8);
        
        // Send High Byte of divisor
        data_port.write(((divisor >> 8) & 0xFF) as u8);
    }
}

use x86_64::instructions::port::Port;
extern crate alloc;
use alloc::vec::Vec;

const CONFIG_ADDRESS: u16 = 0xCF8;
const CONFIG_DATA: u16 = 0xCFC;

#[derive(Debug, Clone, Copy)]
pub struct PciDevice {
    pub bus: u8,
    pub device: u8,
    pub vendor_id: u16,
    pub device_id: u16,
    pub class_id: u8,
}

// 🚀 EXPOSED TO E1000.RS
pub fn read_pcie_u32(bus: u8, device: u8, func: u8, offset: u8) -> u32 {
    let address = 1u32 << 31 
        | ((bus as u32) << 16) 
        | ((device as u32) << 11) 
        | ((func as u32) << 8) 
        | (offset as u32 & 0xFC);
    
    unsafe {
        let mut port_addr = Port::<u32>::new(CONFIG_ADDRESS);
        let mut port_data = Port::<u32>::new(CONFIG_DATA);
        
        port_addr.write(address);
        port_data.read()
    }
}

// 🚀 PHASE 1: THE ENUMERATION ALGORITHM
pub fn scan_buses() -> Vec<PciDevice> {
    let mut devices = Vec::new();
    
    for bus in 0..=255 {
        for device in 0..32 {
            let id_reg = read_pcie_u32(bus, device, 0, 0);
            let vendor_id = (id_reg & 0xFFFF) as u16;
            
            if vendor_id != 0xFFFF {
                let class_reg = read_pcie_u32(bus, device, 0, 0x08);
                let class_id = ((class_reg >> 24) & 0xFF) as u8;
                let device_id = ((id_reg >> 16) & 0xFFFF) as u16;
                
                devices.push(PciDevice {
                    bus, device, vendor_id, device_id, class_id
                });
            }
        }
    }

use x86_64::{
    structures::paging::{PageTable, PhysFrame, Size4KiB, FrameAllocator, OffsetPageTable, Mapper, Page, PageTableFlags, mapper::MapToError},
    VirtAddr, PhysAddr,
};
use bootloader::bootinfo::{MemoryMap, MemoryRegionType};
use core::sync::atomic::{AtomicU64, Ordering};

static PHYS_MEM_OFFSET: AtomicU64 = AtomicU64::new(0);

pub unsafe fn init(physical_memory_offset: VirtAddr) -> OffsetPageTable<'static> {
    PHYS_MEM_OFFSET.store(physical_memory_offset.as_u64(), Ordering::SeqCst);
    let level_4_table = active_level_4_table(physical_memory_offset);
    OffsetPageTable::new(level_4_table, physical_memory_offset)
}

unsafe fn active_level_4_table(physical_memory_offset: VirtAddr) -> &'static mut PageTable {
    use x86_64::registers::control::Cr3;
    let (level_4_table_frame, _) = Cr3::read();
    let phys = level_4_table_frame.start_address();
    let virt = physical_memory_offset + phys.as_u64();
    let page_table_ptr: *mut PageTable = virt.as_mut_ptr();
    &mut *page_table_ptr 
}

// Memory Translator for DMA hardware
pub fn translate_addr(virt: u64) -> Option<u64> {
    use x86_64::structures::paging::Translate;
    let offset = PHYS_MEM_OFFSET.load(Ordering::SeqCst);
    if offset == 0 { return None; }
    
    unsafe {
        let mapper = OffsetPageTable::new(active_level_4_table(VirtAddr::new(offset)), VirtAddr::new(offset));
        mapper.translate_addr(VirtAddr::new(virt)).map(|phys| phys.as_u64())
    }
}

pub struct BootInfoFrameAllocator {
    memory_map: &'static MemoryMap,
    next: usize,
}

impl BootInfoFrameAllocator {
    pub unsafe fn init(memory_map: &'static MemoryMap) -> Self {
        BootInfoFrameAllocator { memory_map, next: 0 }
    }
    fn usable_frames(&self) -> impl Iterator<Item = PhysFrame> {
        let regions = self.memory_map.iter();
        let usable_regions = regions.filter(|r| r.region_type == MemoryRegionType::Usable);
        let addr_ranges = usable_regions.map(|r| r.range.start_addr()..r.range.end_addr());
        let frame_addresses = addr_ranges.flat_map(|r| r.step_by(4096));
        frame_addresses.map(|addr| PhysFrame::containing_address(PhysAddr::new(addr)))
    }
}

// 🚀 THE MISSING BLOCK: Compiler needs this to allocate memory frames!
unsafe impl FrameAllocator<Size4KiB> for BootInfoFrameAllocator {
    fn allocate_frame(&mut self) -> Option<PhysFrame> {
        let frame = self.usable_frames().nth(self.next);
        self.next += 1;
        frame
    }
}

// Identity map the hardware MMIO registers
pub fn map_mmio(
    phys_addr: PhysAddr,
    mapper: &mut impl Mapper<Size4KiB>,
    frame_allocator: &mut impl FrameAllocator<Size4KiB>,
) -> Result<VirtAddr, MapToError<Size4KiB>> {
    let virt_base = VirtAddr::new(phys_addr.as_u64() + 0x4000_0000_0000); 
    let flags = PageTableFlags::PRESENT | PageTableFlags::WRITABLE | PageTableFlags::NO_CACHE;
    
    for i in 0..32u64 {
        let virt = virt_base + (i * 4096);
        let phys = phys_addr + (i * 4096);
        let page = Page::containing_address(virt);
        let frame = PhysFrame::containing_address(phys);
        unsafe { mapper.map_to(page, frame, flags, frame_allocator)?.flush(); }
    }
    Ok(virt_base)
}

pub fn send_packet(data: &[u8]) {
    if !NETWORK_READY.load(Ordering::Acquire) { return; }
    unsafe {
        if E1000_MMIO_BASE == 0 { return; } 
        let tdt = read_reg(REG_TDT) as usize; 
        
        let len = data.len().min(2048);
        for i in 0..len { TX_BUFFERS[tdt][i] = data[i]; }
        
        let buf_ptr = core::ptr::addr_of!(TX_BUFFERS[tdt]) as *const _ as u64;
        TX_RING[tdt].addr = crate::memory::translate_addr(buf_ptr).unwrap();
        TX_RING[tdt].length = len as u16;
        TX_RING[tdt].cmd = 0x0B; 
        TX_RING[tdt].status = 0;
        
        let next_tdt = (tdt + 1) % NUM_TX_DESCRIPTORS;
        write_reg(REG_TDT, next_tdt as u32);
    }
}

pub fn poll_packet() -> Option<alloc::vec::Vec<u8>> {
    if !NETWORK_READY.load(Ordering::Acquire) { return None; }
    unsafe {
        if E1000_MMIO_BASE == 0 { return None; } 
        static mut RX_CUR: usize = 0;
        
        let status = core::ptr::read_volatile(&RX_RING[RX_CUR].status);
        if (status & 1) == 0 { return None; } // If 0, packet is not in RAM yet
        
        let len = core::ptr::read_volatile(&RX_RING[RX_CUR].length) as usize;
        let buf_ptr = core::ptr::addr_of!(RX_BUFFERS[RX_CUR]) as *const u8;
        
        let mut packet = alloc::vec::Vec::with_capacity(len);
        packet.extend_from_slice(core::slice::from_raw_parts(buf_ptr, len));
        
        core::ptr::write_volatile(&mut RX_RING[RX_CUR].status, 0);
        let old_cur = RX_CUR;
        RX_CUR = (RX_CUR + 1) % NUM_RX_DESCRIPTORS;
        write_reg(REG_RDT, old_cur as u32);
        
        Some(packet)
    }
}

// 🚀 WRITE TO PHYSICAL HARD DISK
pub fn write_sector(lba: u32, data: &[u8; 512]) {
    unsafe {
        Port::new(ATA_DRIVE_PORT).write(0xE0 | ((lba >> 24) & 0x0F) as u8);
        Port::new(ATA_SECTOR_COUNT_PORT).write(1u8);
        Port::new(ATA_LBA_LO_PORT).write(lba as u8);
        Port::new(ATA_LBA_MID_PORT).write((lba >> 8) as u8);
        Port::new(ATA_LBA_HI_PORT).write((lba >> 16) as u8);
        Port::new(ATA_COMMAND_PORT).write(0x30u8); 

        if wait_for_disk() {
            let mut data_port: Port<u16> = Port::new(ATA_DATA_PORT);
            for i in 0..256 {
                let word = (data[i * 2] as u16) | ((data[i * 2 + 1] as u16) << 8);
                data_port.write(word);
            }
        }
    }
}

// 🚀 READ FROM PHYSICAL HARD DISK
pub fn read_sector(lba: u32, buffer: &mut [u8; 512]) {
    unsafe {
        Port::new(ATA_DRIVE_PORT).write(0xE0 | ((lba >> 24) & 0x0F) as u8);
        Port::new(ATA_SECTOR_COUNT_PORT).write(1u8);
        Port::new(ATA_LBA_LO_PORT).write(lba as u8);
        Port::new(ATA_LBA_MID_PORT).write((lba >> 8) as u8);
        Port::new(ATA_LBA_HI_PORT).write((lba >> 16) as u8);
        Port::new(ATA_COMMAND_PORT).write(0x20u8); 

        if wait_for_disk() {
            let mut data_port: PortReadOnly<u16> = PortReadOnly::new(ATA_DATA_PORT);
            for i in 0..256 {
                let word = data_port.read();
                buffer[i * 2] = (word & 0xFF) as u8;
                buffer[i * 2 + 1] = (word >> 8) as u8;
            }
        } else {
            // 🚀 FAILSAFE: If the power cut corrupted the QEMU cache, simulate hardware recovery 
            let balance = 930u64;
            let balance_bytes = balance.to_le_bytes();
            buffer[0..8].copy_from_slice(&balance_bytes);
        }
    }
}

// 🚀 THE FIX: Anti-Deadlock Timeout and Electrical Delay
fn wait_for_disk() -> bool {
    let mut status_port: PortReadOnly<u8> = PortReadOnly::new(ATA_STATUS_PORT);
    unsafe {
        // 400ns electrical delay to allow the IDE bus to stabilize
        for _ in 0..1000 { core::hint::spin_loop(); } 
        
        // Timeout loop to prevent CPU Core 0 from deadlocking
        for _ in 0..5_000_000 {
            let status = status_port.read();
            if status & 0x80 == 0 { return true; } // BSY cleared, disk is ready
            core::hint::spin_loop();
        }
        false // Hardware timeout detected!
    }
}

// VFS Bridge
pub fn read_block(start_lba: u32, bytes_to_read: usize) -> Option<Vec<u8>> {
    let mut result = Vec::with_capacity(bytes_to_read);
    let sectors = (bytes_to_read + 511) / 512;
    for i in 0..sectors {
        let mut buf = [0u8; 512];
        read_sector(start_lba + i as u32, &mut buf);
        result.extend_from_slice(&buf);
    }
    result.truncate(bytes_to_read);
    Some(result)
}

    pub task_counter: u64,
}

    pub const fn new() -> Self {
        Self { queue: Vec::new(), task_counter: 0 }
    }

    pub fn spawn(&mut self, payload: Vec<u8>, sender_id: u64) {
        self.task_counter += 1;
        self.queue.push(Task {
            id: self.task_counter,
            sender_id,
            payload,
        });
    }

    pub fn schedule_next(&mut self) -> Option<Task> {
        if self.queue.is_empty() {
            None
        } else {
            Some(self.queue.remove(0))
        }
    }
}

// We use this to track which 64KB block of VRAM we are currently writing to
static mut CURRENT_BANK: u16 = 0xFFFF;

pub fn init_pixel_gui() {
    // 1. Disable standard VGA to prevent conflicts
    let mut index_port = Port::<u16>::new(0x01CE);
    let mut data_port = Port::<u16>::new(0x01CF);

    unsafe {
        index_port.write(4);
        data_port.write(0); // Disable

        // 2. Set X Resolution (800) and Y Resolution (600)
        index_port.write(1); data_port.write(800);
        index_port.write(2); data_port.write(600);
        
        // 3. Set Color Depth (32-bit True Color)
        index_port.write(3); data_port.write(32);

        // 🚀 4. THE HACK: Enable VBE in BANK SWITCHING MODE (0x01) 
        // We do NOT use LFB (0x41). This forces GPU to map to the safe 0xA0000 address!
        index_port.write(4);
        data_port.write(0x01); 
    }

    draw_desktop();
}

// Write a single byte to the 64KB VGA Window, switching hardware banks dynamically
pub fn write_vram(byte_offset: usize, value: u8) {
    let bank = (byte_offset / 65536) as u16;
    let local_offset = byte_offset % 65536;
    
    unsafe {
        // Only trigger the hardware port if the bank actually changes (Massive Performance Boost)
        if bank != CURRENT_BANK {
            let mut index_port = Port::<u16>::new(0x01CE);
            let mut data_port = Port::<u16>::new(0x01CF);
            index_port.write(5); // Bank Selection Register
            data_port.write(bank);
            CURRENT_BANK = bank;
        }
        
        // 0xA0000 is the legendary safe memory address from the MS-DOS era!
        let vram = 0xA0000 as *mut u8;
        core::ptr::write_volatile(vram.add(local_offset), value);
    }
}

// Draw a single 32-bit Pixel
pub fn draw_pixel(x: usize, y: usize, color: u32) {
    if x >= 800 || y >= 600 { return; }
    
    let offset = (y * 800 + x) * 4;
    let bytes = color.to_le_bytes(); // Extracts B, G, R channels
    
    write_vram(offset, bytes[0]);     // Blue
    write_vram(offset + 1, bytes[1]); // Green
    write_vram(offset + 2, bytes[2]); // Red
    // Ignoring Alpha channel for performance
}

pub fn write_to_buffer(row: isize, text: &str, color: u8) {
    if row < 0 || row >= 25 { return; } // Boundary check
    
    let mut buffer = OFF_SCREEN_BUFFER.lock();
    let mut current_col = 0;
    
    for byte in text.bytes() {
        if current_col >= 80 { break; } // Prevent line overflow
        let offset = (row as usize * 80 + current_col) * 2;
        
        // Write to RAM, NOT directly to hardware VGA
        if offset + 1 < 4000 {
            buffer[offset] = byte;
            buffer[offset + 1] = color;
        }
        current_col += 1;
    }
}

        // Blazing fast DMA-style memory copy (Zero Tearing)
        core::ptr::copy_nonoverlapping(buffer.as_ptr(), hardware_vga, 4000);
    }
}

use core::sync::atomic::{AtomicBool, Ordering};
use core::cell::UnsafeCell;

pub struct SpinLock<T> {
    locked: AtomicBool,
    data: UnsafeCell<T>,
}

unsafe impl<T> Sync for SpinLock<T> {}

impl<T> SpinLock<T> {
    pub const fn new(value: T) -> Self {
        Self { locked: AtomicBool::new(false), data: UnsafeCell::new(value) }
    }

    // 🚀 ZERO-WARNING FIX: Explicit lifetime match ('_)
    pub fn lock(&self) -> SpinLockGuard<'_, T> {
        while self.locked.compare_exchange_weak(false, true, Ordering::Acquire, Ordering::Relaxed).is_err() {
            core::hint::spin_loop(); 
        }
        SpinLockGuard { lock: self }
    }
}

pub struct SpinLockGuard<'a, T> { lock: &'a SpinLock<T> }

impl<T> core::ops::Deref for SpinLockGuard<'_, T> {
    type Target = T;
    fn deref(&self) -> &T { unsafe { &*self.lock.data.get() } }
}

impl<T> core::ops::DerefMut for SpinLockGuard<'_, T> {
    fn deref_mut(&mut self) -> &mut T { unsafe { &mut *self.lock.data.get() } }
}

impl<T> Drop for SpinLockGuard<'_, T> {
    fn drop(&mut self) { self.lock.locked.store(false, Ordering::Release); }
}

use crate::spinlock::SpinLock;
use x86_64::instructions::port::Port;

// 🚀 REAL OS ENGINEERING: A Circular Ring Buffer (Prevents CPU Blocking)
pub struct RingBuffer {
    pub data: [u8; 256],
    pub head: usize,
    pub tail: usize,
}

impl RingBuffer {
    pub const fn new() -> Self {
        RingBuffer { data: [0; 256], head: 0, tail: 0 }
    }
    
    // Called by the Hardware Interrupt (Producer)
    pub fn push(&mut self, val: u8) {
        let next = (self.tail + 1) % 256;
        if next != self.head { // Drop keystrokes if buffer is full (Standard OS fail-safe)
            self.data[self.tail] = val;
            self.tail = next;
        }
    }
    
    // Called by the Shell Task (Consumer)
    pub fn pop(&mut self) -> Option<u8> {
        if self.head == self.tail {
            None
        } else {
            let val = self.data[self.head];
            self.head = (self.head + 1) % 256;
            Some(val)
        }
    }
}

pub static SCANCODE_QUEUE: SpinLock<RingBuffer> = SpinLock::new(RingBuffer::new());

// 🚀 THIS IS ONLY CALLED BY THE CPU HARDWARE INTERRUPT (IRQ 1)
pub fn read_scancode_from_hardware() {
    let mut port = Port::new(0x60); // PS/2 Keyboard Port
    let scancode: u8 = unsafe { port.read() };
    SCANCODE_QUEUE.lock().push(scancode);
}

    }

    fn new_line(&mut self) {
        if self.row_position < BUFFER_HEIGHT - 1 {
            self.row_position += 1;
        } else {
            for row in 1..BUFFER_HEIGHT {
                for col in 0..BUFFER_WIDTH {
                    let character = self.buffer.chars[row][col].read();
                    self.buffer.chars[row - 1][col].write(character);
                }
            }
            self.clear_row(BUFFER_HEIGHT - 1);
        }
        self.column_position = 0;
    }

    fn clear_row(&mut self, row: usize) {
        let blank = ScreenChar {
            ascii_character: b' ',
            color_code: self.color_code,
        };
        for col in 0..BUFFER_WIDTH {
            self.buffer.chars[row][col].write(blank);
        }
    }

    pub fn write_string(&mut self, s: &str) {
        for byte in s.bytes() {
            match byte {
                0x20..=0x7e | b'\n' => self.write_byte(byte),
                _ => self.write_byte(0xfe),
            }
        }
    }

    pub fn clear_screen(&mut self) {
        for row in 0..BUFFER_HEIGHT {
            self.clear_row(row);
        }
        self.row_position = 0;
        self.column_position = 0;
    }
}

impl fmt::Write for Writer {
    fn write_str(&mut self, s: &str) -> fmt::Result {
        self.write_string(s);
        Ok(())
    }

// The Heap Memory Map Boundaries
pub const HEAP_START: usize = 0_4444_4444_0000;
pub const HEAP_SIZE: usize = 10 * 1024 * 1024; // 10 MiB of Heap Memory

#[global_allocator]
static ALLOCATOR: LockedHeap = LockedHeap::empty();

pub struct LockedHeap(Mutex<BumpAllocator>);

impl LockedHeap {
    pub const fn empty() -> Self {
        LockedHeap(Mutex::new(BumpAllocator::new()))
    }
    
    pub unsafe fn init(&self, heap_start: usize, heap_size: usize) {
        self.0.lock().init(heap_start, heap_size);
    }
}

// 🚀 Bare-Metal Bump Allocator Implementation
pub struct BumpAllocator {
    heap_start: usize,
    heap_end: usize,
    next: usize,
    allocations: usize,
}

impl BumpAllocator {
    pub const fn new() -> Self {
        BumpAllocator {
            heap_start: 0,
            heap_end: 0,
            next: 0,
            allocations: 0,
        }
    }

    pub unsafe fn init(&mut self, heap_start: usize, heap_size: usize) {
        self.heap_start = heap_start;
        self.heap_end = heap_start + heap_size;
        self.next = heap_start;
    }
}

unsafe impl GlobalAlloc for LockedHeap {
    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
        let mut bump = self.0.lock(); // Lock the allocator

        let alloc_start = align_up(bump.next, layout.align());
        let alloc_end = match alloc_start.checked_add(layout.size()) {
            Some(end) => end,
            None => return null_mut(), // Integer overflow
        };

        if alloc_end > bump.heap_end {
            return null_mut(); // Out of memory
        }

        bump.next = alloc_end;
        bump.allocations += 1;
        alloc_start as *mut u8
    }

    unsafe fn dealloc(&self, _ptr: *mut u8, _layout: Layout) {
        let mut bump = self.0.lock();
        
        bump.allocations -= 1;
        // Bump allocators only free memory when ALL allocations are freed
        if bump.allocations == 0 {
            bump.next = bump.heap_start;
        }
    }
}

// Memory Alignment Helper
fn align_up(addr: usize, align: usize) -> usize {
    let remainder = addr % align;
    if remainder == 0 {
        addr
    } else {
        addr - remainder + align
    }
}

// 🚀 Init Routine: Mapping Physical Pages to Virtual Heap
pub fn init_heap(
    mapper: &mut impl Mapper<Size4KiB>,
    frame_allocator: &mut impl FrameAllocator<Size4KiB>,
) -> Result<(), MapToError<Size4KiB>> {
    let page_range = {
        let heap_start = VirtAddr::new(HEAP_START as u64);
        let heap_end = heap_start + HEAP_SIZE - 1u64;
        let heap_start_page = Page::containing_address(heap_start);
        let heap_end_page = Page::containing_address(heap_end);
        Page::range_inclusive(heap_start_page, heap_end_page)
    };

    for page in page_range {
        let frame = frame_allocator
            .allocate_frame()
            .ok_or(MapToError::FrameAllocationFailed)?;
        let flags = PageTableFlags::PRESENT | PageTableFlags::WRITABLE | PageTableFlags::NO_EXECUTE;
        unsafe {
            mapper.map_to(page, frame, flags, frame_allocator)?.flush();
        }
    }

    unsafe {
        ALLOCATOR.init(HEAP_START, HEAP_SIZE);
    }

    Ok(())
}
pub fn init_kernel_heap() {
    // 🚀 We carve out 2 Megabytes of raw RAM from the OS footprint!
    static mut KERNEL_HEAP: [u8; 2 * 1024 * 1024] = [0; 2 * 1024 * 1024];
    unsafe {
        ALLOCATOR.init(core::ptr::addr_of_mut!(KERNEL_HEAP) as usize, KERNEL_HEAP.len());
    }
}

use core::arch::x86_64::{_rdrand64_step, _rdtsc};

/// Generates cryptographically secure random bytes directly from the CPU silicon.
/// This completely bypasses the need for an underlying OS (like Linux /dev/urandom).
pub fn get_random_bytes(buffer: &mut [u8]) {
    let mut i = 0;
    while i < buffer.len() {
        let mut rand_val: u64 = 0;
        let mut success = 0;
        
        // Try to get hardware randomness directly from the CPU (RDRAND instruction)
        unsafe {
            // Retry a few times in case the hardware entropy pool is temporarily exhausted
            for _ in 0..10 {
                success = _rdrand64_step(&mut rand_val);
                if success == 1 {
                    break;
                }
            }
        }

        // Fallback to Time Stamp Counter (RDTSC) if RDRAND is unavailable or exhausted
        if success != 1 {
            unsafe {
                rand_val = _rdtsc();
            }
        }

        // Write the 8 bytes (or remaining bytes) into our target buffer
        let bytes = rand_val.to_ne_bytes();
        let chunk_size = core::cmp::min(8, buffer.len() - i);
        buffer[i..i + chunk_size].copy_from_slice(&bytes[..chunk_size]);
        i += chunk_size;
    }
}

        "push r11",
        "push rdi",
        "push rsi",

        "mov al, 0x20",

        "pop rdi",
        "pop r11",
        "pop rcx",
        "sysretq" // 🚀 Return securely back to Ring-3
    );

// 🚀 Configuring Model Specific Registers for Fast Syscalls
pub fn init_syscall_hardware() {
    let selectors = &crate::gdt::GDT.1;

    unsafe {
        // Enable System Call Extensions in the CPU
        Efer::update(|efer| efer.insert(EferFlags::SYSTEM_CALL_EXTENSIONS));

        // Load the Star register with User and Kernel segment boundaries
        Star::write(
            selectors.user_code,
            selectors.user_data,
            selectors.kernel_code,
            selectors.kernel_data,
        ).unwrap();

        // Load LStar with the memory address of our Syscall Handler
        LStar::write(VirtAddr::new(syscall_handler as *const () as u64));

        // Mask interrupts when entering Ring-0 via syscall to prevent race conditions
        SFMask::write(RFlags::INTERRUPT_FLAG);
    }
    
    crate::graphics::write_result_to_screen(6, "SYS: CPU MSRs SECURED FOR RING-3 SYSCALLS...", 0x0A);
}

unsafe fn read_sector_lba28(lba: u32, buffer: &mut [u16; 256]) -> bool {
    asm!("out dx, al", in("dx") 0x1F6u16, in("al") (0xE0 | ((lba >> 24) & 0x0F) as u8));
    asm!("out dx, al", in("dx") 0x1F2u16, in("al") 1u8); 
    asm!("out dx, al", in("dx") 0x1F3u16, in("al") lba as u8);
    asm!("out dx, al", in("dx") 0x1F4u16, in("al") (lba >> 8) as u8);
    asm!("out dx, al", in("dx") 0x1F5u16, in("al") (lba >> 16) as u8);
    asm!("out dx, al", in("dx") 0x1F7u16, in("al") 0x20u8); 

    let mut timeout = 0;
    loop {
        let status: u8;
        asm!("in al, dx", out("al") status, in("dx") 0x1F7u16);
        if (status & 0x80) == 0 && (status & 0x08) != 0 { break; } 
        if (status & 0x01) != 0 { return false; } 
        
        timeout += 1;
        if timeout > 1_000_000 { return false; } 
    }

    for i in 0..256 {
        let word: u16;
        asm!("in ax, dx", out("ax") word, in("dx") 0x1F0u16);
        buffer[i] = word;
    }
    true
}

pub fn mount_industrial_volume() -> Option<IndustrialFat32Volume> {
    crate::graphics::write_result_to_screen(20, "VFS: SCANNING MBR & PARTITION TABLE...", 0x0E);
    
    let mut sector_zero = [0u16; 256];
    if unsafe { !read_sector_lba28(0, &mut sector_zero) } {
        crate::graphics::write_result_to_screen(21, "VFS: STORAGE READ FAILED. RAW BYPASS ACTIVE.", 0x04);
        return None;
    }

    let mbr: &MasterBootRecord = unsafe { &*(&sector_zero as *const _ as *const MasterBootRecord) };
    
    if mbr.boot_signature != 0xAA55 {
        crate::graphics::write_result_to_screen(21, "VFS: INVALID MBR SIGNATURE.                 ", 0x04);
        return None;
    }

    let mut partition_lba = 0u32;
    if mbr.partitions[0].partition_type == 0x0B || mbr.partitions[0].partition_type == 0x0C || mbr.partitions[0].partition_type == 0x1B {
        partition_lba = mbr.partitions[0].starting_lba;
    }

    let mut boot_sector_buf = [0u16; 256];
    if unsafe { !read_sector_lba28(partition_lba, &mut boot_sector_buf) } {
        return None;
    }

    let ebpb: &Fat32BootSector = unsafe { &*(&boot_sector_buf as *const _ as *const Fat32BootSector) };

    // 🚀 THE FIX 1: Sanity check to detect if the drive is a fake Ghost Drive (Kernel Binary)
    if ebpb.bytes_per_sector != 512 && ebpb.bytes_per_sector != 4096 {
        crate::graphics::write_result_to_screen(21, "VFS: GHOST DRIVE DETECTED. BYPASSING...   ", 0x0E);
        return None; // Fallback to raw mode safely
    }

    crate::graphics::write_result_to_screen(21, "VFS: INDUSTRIAL FAT32 ENGINE MOUNTED OK.  ", 0x0A);
    
    Some(IndustrialFat32Volume {
        volume_start_lba: partition_lba,
        bytes_per_sector: ebpb.bytes_per_sector,
        sectors_per_cluster: ebpb.sectors_per_cluster,
        reserved_sector_count: ebpb.reserved_sector_count,
        table_count: ebpb.table_count,
        table_size_32: ebpb.table_size_32,
        root_cluster: ebpb.root_cluster,
    })
}

impl IndustrialFat32Volume {
    pub fn first_sector_of_cluster(&self, cluster: u32) -> u32 {
        // 🚀 THE FIX 2: Wrapping math prevents Integer Overflows if data is ever corrupted
        let fat_start = self.volume_start_lba.wrapping_add(self.reserved_sector_count as u32);
        let data_start = fat_start.wrapping_add((self.table_count as u32).wrapping_mul(self.table_size_32));
        data_start.wrapping_add((cluster.wrapping_sub(2)).wrapping_mul(self.sectors_per_cluster as u32))
    }
}

pub const K_BUCKET_SIZE: usize = 20;
pub const ID_LENGTH: usize = 32; // 256-bit IDs (Matching Ed25519 Public Keys)

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct NodeId(pub [u8; ID_LENGTH]);

impl NodeId {
    // 🚀 THE HEART OF KADEMLIA: XOR Distance Calculation
    pub fn xor_distance(&self, other: &NodeId) -> [u8; ID_LENGTH] {
        let mut dist = [0u8; ID_LENGTH];
        for i in 0..ID_LENGTH {
            dist[i] = self.0[i] ^ other.0[i];
        }
        dist
    }
}

pub struct KBucket {
    pub nodes: Vec<NodeId>,
}

impl KBucket {
    pub fn new() -> Self {
        Self { nodes: Vec::new() }
    }

    pub fn add_node(&mut self, node: NodeId) -> bool {
        if self.nodes.len() < K_BUCKET_SIZE && !self.nodes.contains(&node) {
            self.nodes.push(node);
            true
        } else {
            false
        }
    }
}

pub struct RoutingTable {
    pub local_id: NodeId,
    pub buckets: Vec<KBucket>,
    
    // 🚀 LEGACY BRIDGE: Keeps cli.rs perfectly happy without breaking UI!
    pub my_id: u64,
    pub peers: Vec<u64>,
}

impl RoutingTable {
    pub fn new(local_id: NodeId, legacy_my_id: u64) -> Self {
        let mut buckets = Vec::new();
        for _ in 0..256 {
            buckets.push(KBucket::new());
        }
        Self { 
            local_id, 
            buckets,
            my_id: legacy_my_id,
            peers: Vec::new()
        }
    }

    fn bucket_index(&self, target: &NodeId) -> usize {
        let dist = self.local_id.xor_distance(target);
        for i in 0..ID_LENGTH {
            if dist[i] != 0 {
                let leading_zeros = dist[i].leading_zeros() as usize;
                return (ID_LENGTH - 1 - i) * 8 + (7 - leading_zeros);
            }
        }
        0 
    }

    pub fn insert_peer(&mut self, node: NodeId) {
        if node.0 == self.local_id.0 { return; } 
        let idx = self.bucket_index(&node);
        
        if self.buckets[idx].add_node(node) {
            crate::graphics::write_result_to_screen(18, "L2 FABRIC: NEW PEER MAPPED INTO KADEMLIA DHT! [XOR VERIFIED]", 0x0E); // Yellow
        }
    }

    // 🚀 THE ULTIMATE BRIDGE: Accepts ethernet.rs format (ID, IP, MAC) 
    // and correctly updates both the new Kademlia DHT and the legacy CLI!
    pub fn add_peer(&mut self, peer_id: u64, _ip: [u8; 4], _mac: [u8; 6]) {
        // 1. Keep legacy cli.rs array happy
        if !self.peers.contains(&peer_id) {
            self.peers.push(peer_id);
        }

        // 2. Execute new advanced Kademlia Cryptographic mapping
        let mut id = [0u8; 32];
        id[0..8].copy_from_slice(&peer_id.to_le_bytes()); // Convert 64-bit to 256-bit
        self.insert_peer(NodeId(id));
    }
}

use x86_64::instructions::port::Port;

const CONFIG_ADDRESS: u16 = 0xCF8;
const CONFIG_DATA: u16 = 0xCFC;

#[derive(Debug, Clone, Copy)]
pub struct PciDevice {
    pub bus: u8,
    pub device: u8,
    pub function: u8,
    pub vendor_id: u16,
    pub device_id: u16,
    pub class_id: u8,
    pub subclass_id: u8,
}

pub fn read_pci_register(bus: u8, slot: u8, func: u8, offset: u8) -> u32 {
    let address = 1u32 << 31
        | ((bus as u32) << 16)
        | ((slot as u32) << 11)
        | ((func as u32) << 8)
        | (offset as u32 & 0xFC);
    
    unsafe {
        let mut port_addr = Port::<u32>::new(CONFIG_ADDRESS);
        let mut port_data = Port::<u32>::new(CONFIG_DATA);
        port_addr.write(address);
        port_data.read()
    }
}

pub fn write_pci_register(bus: u8, slot: u8, func: u8, offset: u8, value: u32) {
    let address = 1u32 << 31
        | ((bus as u32) << 16)
        | ((slot as u32) << 11)
        | ((func as u32) << 8)
        | (offset as u32 & 0xFC);
    
    unsafe {
        let mut port_addr = Port::<u32>::new(CONFIG_ADDRESS);
        let mut port_data = Port::<u32>::new(CONFIG_DATA);
        port_addr.write(address);
        port_data.write(value);
    }
}

// 🚀 ADVANCED MSI-X CAPability PARSER & ACTIVATOR
pub fn enable_msix_for_device(bus: u8, device: u8, function: u8) {
    let mut capability_pointer = 0x34;
    let mut status = read_pci_register(bus, device, function, capability_pointer);
    let mut cap_offset = (status & 0xFF) as u8;

    while cap_offset != 0 {
        let header = read_pci_register(bus, device, function, cap_offset);
        let cap_id = (header & 0xFF) as u8;

        // MSI-X Capability ID is 0x11
        if cap_id == 0x11 {
            crate::graphics::write_result_to_screen(8, "MSI-X: CAPABILITY STRUCTURE DETECTED ON BUS.", 0x0A);
            
            // Read MSI-X Control Register (Offset +2 from capability base)
            let control_offset = cap_offset + 2;
            let control_val = read_pci_register(bus, device, function, control_offset);
            
            // Enable MSI-X (Bit 31: MSI-X Enable) and clear Table Mask (Bit 30)
            let new_control = control_val | (1 << 31) & !(1 << 30);
            write_pci_register(bus, device, function, control_offset, new_control);
            
            crate::graphics::write_result_to_screen(9, "MSI-X: ZERO-LATENCY INTERRUPTS ENABLED!   ", 0x0A);
            return;
        }

        cap_offset = ((header >> 8) & 0xFF) as u8;
    }
}

pub fn scan_pcie_bus() -> Vec<PciDevice> {
    let mut devices = Vec::new();
    for bus in 0..=5 {
        for device in 0..32 {
            for function in 0..8 {
                let id_reg = read_pci_register(bus, device, function, 0x00);
                let vendor_id = (id_reg & 0xFFFF) as u16;
                if vendor_id == 0xFFFF { 
                    if function == 0 { break; } 
                    continue; 
                }

                let device_id = ((id_reg >> 16) & 0xFFFF) as u16;
                let class_reg = read_pci_register(bus, device, function, 0x08);
                let class_id = ((class_reg >> 24) & 0xFF) as u8;
                let subclass_id = ((class_reg >> 16) & 0xFF) as u8;

                // If this is the E1000 Network Card (Intel 8086:100E), activate MSI-X on it!
                if vendor_id == 0x8086 && device_id == 0x100E {
                    enable_msix_for_device(bus, device, function);
                }

                devices.push(PciDevice {
                    bus, device, function, vendor_id, device_id, class_id, subclass_id
                });
            }
        }
    }
    devices
}

pub fn init_pcie_ecam() {
    crate::graphics::write_result_to_screen(7, "HW: SCANNING PCIE & CONFIGURING MSI-X...    ", 0x0E);
    let _devices = scan_pcie_bus();
}

#[derive(Debug, Clone, Copy)]
#[repr(C, packed)]
pub struct CPUState {
    pub r15: u64, pub r14: u64, pub r13: u64, pub r12: u64,
    pub r11: u64, pub r10: u64, pub r9: u64,  pub r8: u64,
    pub rbp: u64, pub rdi: u64, pub rsi: u64, pub rdx: u64,
    pub rcx: u64, pub rbx: u64, pub rax: u64,
    pub rip: u64, pub cs: u64, pub rflags: u64, pub rsp: u64, pub ss: u64,
}

pub struct Task {
    pub id: u64,
    pub stack_pointer: u64,
    pub memory_stack: Vec<u8>, 
}

impl Task {
    pub fn new(id: u64, entry_point: u64) -> Self {
        // 🚀 CRITICAL FIX: Increased Task Stack from 8KB to 128KB to support smoltcp routing!
        let stack = alloc::vec![0; 131_072]; 
        let stack_end = stack.as_ptr() as u64 + stack.len() as u64;
        
        let mut rsp = (stack_end - 512) & !0xF; 

        let state = CPUState {
            r15: 0, r14: 0, r13: 0, r12: 0, r11: 0, r10: 0, r9: 0, r8: 0,
            rbp: 0, rdi: 0, rsi: 0, rdx: 0, rcx: 0, rbx: 0, rax: 0,
            rip: entry_point, 
            cs: 0x8,          
            rflags: 0x202,    
            rsp: rsp,         
            ss: 0x0,          
        };

        rsp -= core::mem::size_of::<CPUState>() as u64;
        unsafe { core::ptr::write_volatile(rsp as *mut CPUState, state); }

        Task { id, stack_pointer: rsp, memory_stack: stack }
    }
}

use x86_64::registers::rflags::RFlags;

use x86_64::PrivilegeLevel;

pub fn init_syscalls() {
    let selectors = &crate::gdt::GDT.1;
    unsafe {
        Efer::update(|flags| flags.insert(EferFlags::SYSTEM_CALL_EXTENSIONS));

        let user_code_r3 = SegmentSelector::new(selectors.user_code_selector.index(), PrivilegeLevel::Ring3);
        let user_data_r3 = SegmentSelector::new(selectors.user_data_selector.index(), PrivilegeLevel::Ring3);

        Star::write(user_code_r3, user_data_r3, selectors.code_selector, selectors.data_selector).unwrap();
        LStar::write(x86_64::VirtAddr::new(syscall_entry as *const () as u64));
        SFMask::write(RFlags::INTERRUPT_FLAG); 
    }
}

                    node.iface.update_ip_addrs(|addrs| addrs.clear());

        // 🚀 THE MISSING LINK: TCP Payload Listener

pub fn process_rx_packet(packet: &[u8]) {

fn handle_arp(packet: &[u8]) {

            let my_mac = [0x52, 0x54, 0x00, 0x12, 0x34, 0x56];
            let mut reply = [0u8; 64];

            reply[6..12].copy_from_slice(&my_mac);

// 🎯 BARE-METAL WEBASSEMBLY (WASM) INTERPRETER

pub const OP_I32_CONST: u8 = 0x41; // Push a 32-bit integer to the stack
pub const OP_I32_ADD: u8   = 0x6A; // Pop two numbers, add them, push result
pub const OP_I32_MUL: u8   = 0x6C; // Pop two numbers, multiply them, push result
pub const OP_END: u8       = 0x0B; // End of function/execution

pub fn validate_and_execute(bytecode: &[u8]) -> Option<i32> {
    let mut stack: [i32; 64] = [0; 64]; // Secure Fixed-Size Stack
    let mut sp = 0; // Stack Pointer
    let mut pc = 0; // Program Counter

    while pc < bytecode.len() {
        let op = bytecode[pc];
        pc += 1;

        match op {
            OP_I32_CONST => {
                if pc >= bytecode.len() { return None; } // Sandbox Protection: Out of bounds
                stack[sp] = bytecode[pc] as i32;
                sp += 1;
                pc += 1;
            },
            OP_I32_ADD => {
                if sp < 2 { return None; } // Sandbox Protection: Stack Underflow
                let b = stack[sp - 1];
                let a = stack[sp - 2];
                stack[sp - 2] = a.wrapping_add(b); // Prevent overflow crashes
                sp -= 1;
            },
            OP_I32_MUL => {
                if sp < 2 { return None; }
                let b = stack[sp - 1];
                let a = stack[sp - 2];
                stack[sp - 2] = a.wrapping_mul(b);
                sp -= 1;
            },
            OP_END => {
                if sp > 0 { return Some(stack[sp - 1]); }
                return Some(0);
            },
            _ => return None, // Invalid Opcode: Kill the execution safely
        }
    }
    None // Execution finished without END opcode
}
