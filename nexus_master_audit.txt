// nexus-core/src/timer.rs

use x86_64::instructions::port::Port;

// PIT Base Frequency is ~1.193182 MHz
const PIT_DIVIDEND: u32 = 1193182;

pub fn configure_pit(target_hz: u32) {
    let mut command_port: Port<u8> = Port::new(0x43);
    let mut data_port: Port<u8> = Port::new(0x40);

    let divisor = (PIT_DIVIDEND / target_hz) as u16;

    unsafe {
        // Command byte: Channel 0, LSB/MSB access, Mode 2 (Rate Generator), Binary mode
        command_port.write(0x36);
        
        // Send Low Byte of divisor
        data_port.write((divisor & 0xFF) as u8);
        
        // Send High Byte of divisor
        data_port.write(((divisor >> 8) & 0xFF) as u8);
    }
}

extern crate alloc;
use alloc::vec::Vec;
use crate::spinlock::SpinLock;

// 160-bit Kademlia Node ID
#[derive(Clone, Copy, PartialEq, Eq)]
pub struct NodeId(pub [u8; 20]);

#[derive(Clone)]
pub struct PeerNode {
    pub id: NodeId,
    pub ip: [u8; 4],
    pub port: u16,
}

// 🚀 THE SMP FIX: The Global Thread-Safe Routing Table
pub static ROUTING_TABLE: SpinLock<Vec<PeerNode>> = SpinLock::new(Vec::new());

// Mathematical XOR distance calculation
pub fn xor_distance(a: &NodeId, b: &NodeId) -> NodeId {
    let mut result = [0; 20];
    for i in 0..20 {
        result[i] = a.0[i] ^ b.0[i];
    }
    NodeId(result)
}

// Thread-safe peer insertion
pub fn add_peer(new_peer: PeerNode) {
    let mut table = ROUTING_TABLE.lock(); // Hardware lock engaged!
    
    let mut exists = false;
    for peer in table.iter() {
        if peer.id.0 == new_peer.id.0 {
            exists = true;
            break;
        }
    }
    
    if !exists {
        table.push(new_peer);
    }
    // Hardware lock automatically releases here when 'table' goes out of scope
}

// Thread-safe peer counter for the dashboard
pub fn get_peer_count() -> usize {
    let table = ROUTING_TABLE.lock();
    table.len()
}

use x86_64::instructions::port::Port;
extern crate alloc;
use alloc::vec::Vec;

const CONFIG_ADDRESS: u16 = 0xCF8;
const CONFIG_DATA: u16 = 0xCFC;

#[derive(Debug, Clone, Copy)]
pub struct PciDevice {
    pub bus: u8,
    pub device: u8,
    pub vendor_id: u16,
    pub device_id: u16,
    pub class_id: u8,
}

// 🚀 EXPOSED TO E1000.RS
pub fn read_pcie_u32(bus: u8, device: u8, func: u8, offset: u8) -> u32 {
    let address = 1u32 << 31 
        | ((bus as u32) << 16) 
        | ((device as u32) << 11) 
        | ((func as u32) << 8) 
        | (offset as u32 & 0xFC);
    
    unsafe {
        let mut port_addr = Port::<u32>::new(CONFIG_ADDRESS);
        let mut port_data = Port::<u32>::new(CONFIG_DATA);
        
        port_addr.write(address);
        port_data.read()
    }
}

// 🚀 PHASE 1: THE ENUMERATION ALGORITHM
pub fn scan_buses() -> Vec<PciDevice> {
    let mut devices = Vec::new();
    
    for bus in 0..=255 {
        for device in 0..32 {
            let id_reg = read_pcie_u32(bus, device, 0, 0);
            let vendor_id = (id_reg & 0xFFFF) as u16;
            
            if vendor_id != 0xFFFF {
                let class_reg = read_pcie_u32(bus, device, 0, 0x08);
                let class_id = ((class_reg >> 24) & 0xFF) as u8;
                let device_id = ((id_reg >> 16) & 0xFFFF) as u16;
                
                devices.push(PciDevice {
                    bus, device, vendor_id, device_id, class_id
                });
            }
        }
    }
    devices
}

// 🚀 LEGACY COMPATIBILITY
pub fn scan_bus_and_get_e1000_io_base() -> Option<(u32, u8, u8)> {
    for bus in 0..=255 {
        for device in 0..32 {

            }
        }
    }
    None
}

    frame[6..12].copy_from_slice(&[0x52, 0x54, 0x00, 0x12, 0x34, 0x56]);
    frame[12] = 0x88; frame[13] = 0xB5;
    frame[14] = RPC_RESULT;
    
    // 🚀 UPGRADED: 32-bit full payload serialization for large hashes
    frame[15] = (result & 0xFF) as u8;
    frame[16] = ((result >> 8) & 0xFF) as u8;
    frame[17] = ((result >> 16) & 0xFF) as u8;
    frame[18] = ((result >> 24) & 0xFF) as u8;
    
    crate::e1000::send_packet(&frame);
}

            RPC_RESULT => {

use x86_64::{
    structures::paging::{PageTable, PhysFrame, Size4KiB, FrameAllocator, OffsetPageTable, Mapper, Page, PageTableFlags, mapper::MapToError},
    VirtAddr, PhysAddr,
};
use bootloader::bootinfo::{MemoryMap, MemoryRegionType};
use core::sync::atomic::{AtomicU64, Ordering};

static PHYS_MEM_OFFSET: AtomicU64 = AtomicU64::new(0);

pub unsafe fn init(physical_memory_offset: VirtAddr) -> OffsetPageTable<'static> {
    PHYS_MEM_OFFSET.store(physical_memory_offset.as_u64(), Ordering::SeqCst);
    let level_4_table = active_level_4_table(physical_memory_offset);
    OffsetPageTable::new(level_4_table, physical_memory_offset)
}

unsafe fn active_level_4_table(physical_memory_offset: VirtAddr) -> &'static mut PageTable {
    use x86_64::registers::control::Cr3;
    let (level_4_table_frame, _) = Cr3::read();
    let phys = level_4_table_frame.start_address();
    let virt = physical_memory_offset + phys.as_u64();
    let page_table_ptr: *mut PageTable = virt.as_mut_ptr();
    &mut *page_table_ptr 
}

// Memory Translator for DMA hardware
pub fn translate_addr(virt: u64) -> Option<u64> {
    use x86_64::structures::paging::Translate;
    let offset = PHYS_MEM_OFFSET.load(Ordering::SeqCst);
    if offset == 0 { return None; }
    
    unsafe {
        let mapper = OffsetPageTable::new(active_level_4_table(VirtAddr::new(offset)), VirtAddr::new(offset));
        mapper.translate_addr(VirtAddr::new(virt)).map(|phys| phys.as_u64())
    }
}

pub struct BootInfoFrameAllocator {
    memory_map: &'static MemoryMap,
    next: usize,
}

impl BootInfoFrameAllocator {
    pub unsafe fn init(memory_map: &'static MemoryMap) -> Self {
        BootInfoFrameAllocator { memory_map, next: 0 }
    }
    fn usable_frames(&self) -> impl Iterator<Item = PhysFrame> {
        let regions = self.memory_map.iter();
        let usable_regions = regions.filter(|r| r.region_type == MemoryRegionType::Usable);
        let addr_ranges = usable_regions.map(|r| r.range.start_addr()..r.range.end_addr());
        let frame_addresses = addr_ranges.flat_map(|r| r.step_by(4096));
        frame_addresses.map(|addr| PhysFrame::containing_address(PhysAddr::new(addr)))
    }
}

// 🚀 THE MISSING BLOCK: Compiler needs this to allocate memory frames!
unsafe impl FrameAllocator<Size4KiB> for BootInfoFrameAllocator {
    fn allocate_frame(&mut self) -> Option<PhysFrame> {
        let frame = self.usable_frames().nth(self.next);
        self.next += 1;
        frame
    }
}

// Identity map the hardware MMIO registers
pub fn map_mmio(
    phys_addr: PhysAddr,
    mapper: &mut impl Mapper<Size4KiB>,
    frame_allocator: &mut impl FrameAllocator<Size4KiB>,
) -> Result<VirtAddr, MapToError<Size4KiB>> {
    let virt_base = VirtAddr::new(phys_addr.as_u64() + 0x4000_0000_0000); 
    let flags = PageTableFlags::PRESENT | PageTableFlags::WRITABLE | PageTableFlags::NO_CACHE;
    
    for i in 0..32u64 {
        let virt = virt_base + (i * 4096);
        let phys = phys_addr + (i * 4096);
        let page = Page::containing_address(virt);
        let frame = PhysFrame::containing_address(phys);
        unsafe { mapper.map_to(page, frame, flags, frame_allocator)?.flush(); }
    }
    Ok(virt_base)
}

pub fn send_packet(data: &[u8]) {
    // 🚀 THE FIX: Block transmission if setup isn't finished
    if !NETWORK_READY.load(Ordering::Acquire) { return; }
    
    unsafe {
        if E1000_MMIO_BASE == 0 { return; } 
        let tdt = read_reg(REG_TDT) as usize; 
        
        let len = data.len().min(2048);
        for i in 0..len { TX_BUFFERS[tdt][i] = data[i]; }
        
        let tx_buf_phys = match crate::memory::translate_addr(TX_BUFFERS[tdt].as_ptr() as u64) {
            Some(addr) => addr,
            None => return,
        };
        TX_RING[tdt].addr = tx_buf_phys;
        TX_RING[tdt].length = len as u16;
        TX_RING[tdt].cmd = 0x0B; // EOP | IFCS | RS
        TX_RING[tdt].status = 0;
        
        let next_tdt = (tdt + 1) % NUM_TX_DESCRIPTORS;
        write_reg(REG_TDT, next_tdt as u32); // RING THE DOORBELL
    }
}

pub fn receive_packets() {
    // 🚀 THE FIX: Block background network daemon from polling uninitialized hardware
    if !NETWORK_READY.load(Ordering::Acquire) { return; }
    
    unsafe {
        if E1000_MMIO_BASE == 0 { return; } 
        
        static mut RX_CUR: usize = 0;
        
        loop {
            if RX_CUR >= NUM_RX_DESCRIPTORS { RX_CUR = 0; } // Bounds protection
            
            let status = core::ptr::read_volatile(&RX_RING[RX_CUR].status);
            if (status & 1) == 0 {
                break; 
            }
            
            let len = core::ptr::read_volatile(&RX_RING[RX_CUR].length) as usize;
            
            let buf_ptr = RX_BUFFERS[RX_CUR].as_ptr();
            let packet = core::slice::from_raw_parts(buf_ptr, len);
            
            crate::fabric::process_incoming_frame(packet);
            
            core::ptr::write_volatile(&mut RX_RING[RX_CUR].status, 0);
            
            let old_cur = RX_CUR;
            RX_CUR = (RX_CUR + 1) % NUM_RX_DESCRIPTORS;
            write_reg(REG_RDT, old_cur as u32);
        }
    }
}

const VGA_WIDTH: usize = 80;
const VGA_HEIGHT: usize = 25;

pub fn write_result_to_screen(row: usize, text: &str, color: u8) {
    let _guard = DISPLAY_LOCK.lock(); 
    
    if row >= VGA_HEIGHT { return; }
    
    let vga = unsafe { OFF_SCREEN_BUFFER.as_mut_ptr() };
    let mut col = 0;
    
    // 1. Write the new text
    for byte in text.bytes() {
        if col >= VGA_WIDTH { break; }
        let offset = (row * VGA_WIDTH + col) * 2;
        unsafe {
            ptr::write_volatile(vga.add(offset), byte);
            ptr::write_volatile(vga.add(offset + 1), color);
        }
        col += 1;
    }
    
    // 2. 🚀 THE FIX: Sweep the rest of the row with blank spaces to kill ghost text!
    while col < VGA_WIDTH {
        let offset = (row * VGA_WIDTH + col) * 2;
        unsafe {
            ptr::write_volatile(vga.add(offset), b' ');
            ptr::write_volatile(vga.add(offset + 1), 0x00); // Black background
        }
        col += 1;
    }
    
    // 3. Flush to monitor instantly
    let hardware_vga = 0xb8000 as *mut u8;
    unsafe {
        core::ptr::copy_nonoverlapping(OFF_SCREEN_BUFFER.as_ptr(), hardware_vga, 4000);
    }
}

        Port::new(ATA_SECTOR_COUNT_PORT).write(1u8);
        Port::new(ATA_LBA_LO_PORT).write(lba as u8);
        Port::new(ATA_LBA_MID_PORT).write((lba >> 8) as u8);
        Port::new(ATA_LBA_HI_PORT).write((lba >> 16) as u8);
        Port::new(ATA_COMMAND_PORT).write(0x30u8); 

        if wait_for_disk() {

            for i in 0..256 {
                let word = (data[i * 2] as u16) | ((data[i * 2 + 1] as u16) << 8);
                data_port.write(word);
            }
        }
    }
}

            for i in 0..256 {
                let word = data_port.read();
                buffer[i * 2] = (word & 0xFF) as u8;

            let balance = 930u64;

// 🚀 THE FIX: Anti-Deadlock Timeout and Electrical Delay
fn wait_for_disk() -> bool {
    let mut status_port: PortReadOnly<u8> = PortReadOnly::new(ATA_STATUS_PORT);
    unsafe {
        // 400ns electrical delay to allow the IDE bus to stabilize
        for _ in 0..1000 { core::hint::spin_loop(); } 
        
        // Timeout loop to prevent CPU Core 0 from deadlocking
        for _ in 0..5_000_000 {
            let status = status_port.read();
            if status & 0x80 == 0 { return true; } // BSY cleared, disk is ready
            core::hint::spin_loop();
        }
        false // Hardware timeout detected!
    }
}

        if self.queue.is_empty() {
            None
        } else {
            Some(self.queue.remove(0))
        }
    }
}

static mut CURRENT_BANK: u16 = 0xFFFF;

pub fn init_pixel_gui() {

    let mut index_port = Port::<u16>::new(0x01CE);
    let mut data_port = Port::<u16>::new(0x01CF);

    unsafe {
        index_port.write(4);
        data_port.write(0); // Disable

        // 2. Set X Resolution (800) and Y Resolution (600)
        index_port.write(1); data_port.write(800);
        index_port.write(2); data_port.write(600);
        
        // 3. Set Color Depth (32-bit True Color)
        index_port.write(3); data_port.write(32);

        // 🚀 4. THE HACK: Enable VBE in BANK SWITCHING MODE (0x01) 
        // We do NOT use LFB (0x41). This forces GPU to map to the safe 0xA0000 address!
        index_port.write(4);
        data_port.write(0x01); 
    }

    draw_desktop();
}

// Write a single byte to the 64KB VGA Window, switching hardware banks dynamically
pub fn write_vram(byte_offset: usize, value: u8) {
    let bank = (byte_offset / 65536) as u16;
    let local_offset = byte_offset % 65536;
    
    unsafe {
        // Only trigger the hardware port if the bank actually changes (Massive Performance Boost)
        if bank != CURRENT_BANK {
            let mut index_port = Port::<u16>::new(0x01CE);
            let mut data_port = Port::<u16>::new(0x01CF);
            index_port.write(5); // Bank Selection Register
            data_port.write(bank);
            CURRENT_BANK = bank;
        }
        
        // 0xA0000 is the legendary safe memory address from the MS-DOS era!
        let vram = 0xA0000 as *mut u8;
        core::ptr::write_volatile(vram.add(local_offset), value);
    }
}

}

pub static OFF_SCREEN_BUFFER: SpinLock<[u8; 4000]> = SpinLock::new([0; 4000]);

pub fn write_to_buffer(row: isize, text: &str, color: u8) {
    if row < 0 || row >= 25 { return; } // Boundary check
    
    let mut buffer = OFF_SCREEN_BUFFER.lock();
    let mut current_col = 0;
    
    for byte in text.bytes() {
        if current_col >= 80 { break; } // Prevent line overflow
        let offset = (row as usize * 80 + current_col) * 2;
        
        // Write to RAM, NOT directly to hardware VGA
        if offset + 1 < 4000 {
            buffer[offset] = byte;
            buffer[offset + 1] = color;
        }
        current_col += 1;
    }
}

        // Blazing fast DMA-style memory copy (Zero Tearing)
        core::ptr::copy_nonoverlapping(buffer.as_ptr(), hardware_vga, 4000);
    }
}

// nexus-core/src/spinlock.rs

use core::sync::atomic::{AtomicBool, Ordering};
use core::cell::UnsafeCell;

pub struct SpinLock<T> {
    locked: AtomicBool,
    data: UnsafeCell<T>,
}

unsafe impl<T> Sync for SpinLock<T> {}

impl<T> SpinLock<T> {
    pub const fn new(value: T) -> Self {
        Self {
            locked: AtomicBool::new(false),
            data: UnsafeCell::new(value),
        }
    }

    pub fn lock(&self) -> SpinLockGuard<T> {
        // Spin loop: Wait until 'locked' becomes false, then set it to true.
        // Uses Acquire ordering to ensure memory safety across CPU cores.
        while self.locked.compare_exchange_weak(false, true, Ordering::Acquire, Ordering::Relaxed).is_err() {
            core::hint::spin_loop(); 
        }
        SpinLockGuard { lock: self }
    }
}

pub struct SpinLockGuard<'a, T> {
    lock: &'a SpinLock<T>,
}

impl<T> core::ops::Deref for SpinLockGuard<'_, T> {
    type Target = T;
    fn deref(&self) -> &T {
        unsafe { &*self.lock.data.get() }
    }
}

impl<T> core::ops::DerefMut for SpinLockGuard<'_, T> {
    fn deref_mut(&mut self) -> &mut T {
        unsafe { &mut *self.lock.data.get() }
    }
}

impl<T> Drop for SpinLockGuard<'_, T> {
    fn drop(&mut self) {
        // Release the lock for other CPU cores
        self.lock.locked.store(false, Ordering::Release);
    }
}

// 🚀 REAL OS ENGINEERING: A Circular Ring Buffer (Prevents CPU Blocking)
pub struct RingBuffer {
    pub data: [u8; 256],
    pub head: usize,
    pub tail: usize,
}

impl RingBuffer {
    pub const fn new() -> Self {
        RingBuffer { data: [0; 256], head: 0, tail: 0 }
    }
    
    // Called by the Hardware Interrupt (Producer)
    pub fn push(&mut self, val: u8) {
        let next = (self.tail + 1) % 256;
        if next != self.head { // Drop keystrokes if buffer is full (Standard OS fail-safe)
            self.data[self.tail] = val;
            self.tail = next;
        }
    }
    
    // Called by the Shell Task (Consumer)
    pub fn pop(&mut self) -> Option<u8> {
        if self.head == self.tail {
            None
        } else {
            let val = self.data[self.head];
            self.head = (self.head + 1) % 256;
            Some(val)
        }
    }
}

    let mut port = Port::new(0x60); // PS/2 Keyboard Port
    let scancode: u8 = unsafe { port.read() };
    SCANCODE_QUEUE.lock().push(scancode);
}

                    let character = self.buffer.chars[row][col].read();
                    self.buffer.chars[row - 1][col].write(character);
                }

            }
        }

static mut HEAP_MEMORY: [u8; HEAP_SIZE] = [0; HEAP_SIZE];

// 🚀 2. THE ALLOCATOR LOGIC
pub struct BumpAllocator {
    heap_start: usize,
    heap_end: usize,
    next: usize,
    allocations: usize,
}

impl BumpAllocator {
    pub const fn new() -> Self {
        BumpAllocator { heap_start: 0, heap_end: 0, next: 0, allocations: 0 }
    }
    
    pub unsafe fn init(&mut self, heap_start: usize, heap_size: usize) {
        self.heap_start = heap_start;
        self.heap_end = heap_start + heap_size;
        self.next = heap_start;
    }
}

// Memory alignment helper
fn align_up(addr: usize, align: usize) -> usize {
    (addr + align - 1) & !(align - 1)
}

// 🚀 3. THE GLOBAL ALLOCATOR TRAIT IMPLEMENTATION
pub struct Locked<A> { inner: Mutex<A> }
impl<A> Locked<A> { pub const fn new(inner: A) -> Self { Locked { inner: Mutex::new(inner) } } }

unsafe impl GlobalAlloc for Locked<BumpAllocator> {
    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
        let mut bump = self.inner.lock();
        
        let alloc_start = align_up(bump.next, layout.align());
        let alloc_end = alloc_start.saturating_add(layout.size());

        if alloc_end <= bump.heap_end {
            bump.next = alloc_end;
            bump.allocations += 1;
            alloc_start as *mut u8
        } else {
            null_mut() // Out of memory (OOM) panic trigger
        }
    }

    unsafe fn dealloc(&self, _ptr: *mut u8, _layout: Layout) {
        let mut bump = self.inner.lock();
        bump.allocations -= 1;
        
        // If all memory is freed, completely reset the heap pointer for reuse!
        if bump.allocations == 0 {
            bump.next = bump.heap_start;
        }
    }
}

use core::arch::x86_64::{_rdrand64_step, _rdtsc};

/// Generates cryptographically secure random bytes directly from the CPU silicon.
/// This completely bypasses the need for an underlying OS (like Linux /dev/urandom).
pub fn get_random_bytes(buffer: &mut [u8]) {
    let mut i = 0;
    while i < buffer.len() {
        let mut rand_val: u64 = 0;
        let mut success = 0;
        
        // Try to get hardware randomness directly from the CPU (RDRAND instruction)
        unsafe {
            // Retry a few times in case the hardware entropy pool is temporarily exhausted
            for _ in 0..10 {
                success = _rdrand64_step(&mut rand_val);
                if success == 1 {
                    break;
                }
            }
        }

        // Fallback to Time Stamp Counter (RDTSC) if RDRAND is unavailable or exhausted
        if success != 1 {
            unsafe {
                rand_val = _rdtsc();
            }
        }

        // Write the 8 bytes (or remaining bytes) into our target buffer
        let bytes = rand_val.to_ne_bytes();
        let chunk_size = core::cmp::min(8, buffer.len() - i);
        buffer[i..i + chunk_size].copy_from_slice(&bytes[..chunk_size]);
        i += chunk_size;
    }
}

extern crate alloc;
use alloc::collections::VecDeque;
use spin::Mutex;
use crate::process::Task;

pub static TASK_QUEUE: Mutex<VecDeque<Task>> = Mutex::new(VecDeque::new());
// 🚀 THE GHOST BUG FIX: Track the currently running task separately!
pub static CURRENT_TASK: Mutex<Option<Task>> = Mutex::new(None);

#[no_mangle]
pub extern "C" fn schedule_next_task(current_rsp: u64) -> u64 {
    // Acknowledge PIC interrupt (Port 0x20)
    unsafe { x86_64::instructions::port::Port::<u8>::new(0x20).write(0x20); }

    let mut queue = TASK_QUEUE.lock();
    let mut current_task_lock = CURRENT_TASK.lock();

    if queue.is_empty() {
        return current_rsp; 
    }

    // Save the outgoing task (if any)
    if let Some(mut task) = current_task_lock.take() {
        task.stack_pointer = current_rsp;
        queue.push_back(task);
    }

    // Load the incoming task
    let next_task = queue.pop_front().unwrap();
    let next_rsp = next_task.stack_pointer;
    
    *current_task_lock = Some(next_task);

    next_rsp
}

pub fn add_task(entry_point: fn()) {
    let mut queue = TASK_QUEUE.lock();
    let id = queue.len() as u64 + 1;
    let new_task = Task::new(id, entry_point as u64);
    queue.push_back(new_task);
}

pub const K_BUCKET_SIZE: usize = 20;
pub const ID_LENGTH: usize = 32; // 256-bit IDs (Matching Ed25519 Public Keys)

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct NodeId(pub [u8; ID_LENGTH]);

impl NodeId {
    // 🚀 THE HEART OF KADEMLIA: XOR Distance Calculation
    pub fn xor_distance(&self, other: &NodeId) -> [u8; ID_LENGTH] {
        let mut dist = [0u8; ID_LENGTH];
        for i in 0..ID_LENGTH {
            dist[i] = self.0[i] ^ other.0[i];
        }
        dist
    }
}

pub struct KBucket {
    pub nodes: Vec<NodeId>,
}

impl KBucket {
    pub fn new() -> Self {
        Self { nodes: Vec::new() }
    }

    pub fn add_node(&mut self, node: NodeId) -> bool {
        if self.nodes.len() < K_BUCKET_SIZE && !self.nodes.contains(&node) {
            self.nodes.push(node);
            true
        } else {
            false
        }
    }
}

pub struct RoutingTable {
    pub local_id: NodeId,
    pub buckets: Vec<KBucket>,
    
    // 🚀 LEGACY BRIDGE: Keeps cli.rs perfectly happy without breaking UI!
    pub my_id: u64,
    pub peers: Vec<u64>,
}

impl RoutingTable {
    pub fn new(local_id: NodeId, legacy_my_id: u64) -> Self {
        let mut buckets = Vec::new();
        for _ in 0..256 {
            buckets.push(KBucket::new());
        }
        Self { 
            local_id, 
            buckets,
            my_id: legacy_my_id,
            peers: Vec::new()
        }
    }

    fn bucket_index(&self, target: &NodeId) -> usize {
        let dist = self.local_id.xor_distance(target);
        for i in 0..ID_LENGTH {
            if dist[i] != 0 {
                let leading_zeros = dist[i].leading_zeros() as usize;
                return (ID_LENGTH - 1 - i) * 8 + (7 - leading_zeros);
            }
        }
        0 
    }

    pub fn insert_peer(&mut self, node: NodeId) {
        if node.0 == self.local_id.0 { return; } 
        let idx = self.bucket_index(&node);
        
        if self.buckets[idx].add_node(node) {
            crate::graphics::write_result_to_screen(18, "L2 FABRIC: NEW PEER MAPPED INTO KADEMLIA DHT! [XOR VERIFIED]", 0x0E); // Yellow
        }
    }

    // and correctly updates both the new Kademlia DHT and the legacy CLI!

        id[0..8].copy_from_slice(&peer_id.to_le_bytes()); // Convert 64-bit to 256-bit

    }

fn handle_arp(packet: &[u8]) {

fn handle_ipv4(packet: &[u8]) {
    if packet.len() < 34 { return; }
    let protocol = packet[23];

    if protocol == 1 { 
        let icmp_type = packet[34];
        if icmp_type == 8 { 
            host_log("   ⚡ [ ICMP ] PING Received! Sending Echo Reply...");
            let mut reply = [0u8; 128]; 
            let len = packet.len();
            if len > 128 { return; }
            reply[0..len].copy_from_slice(&packet[0..len]);
            for i in 0..6 { reply[i] = packet[i+6]; reply[i+6] = packet[i]; }
            for i in 0..4 { reply[26+i] = packet[30+i]; reply[30+i] = packet[26+i]; }
            reply[34] = 0; reply[36] = 0; reply[37] = 0; 
            let mut sum: u32 = 0;
            for i in (34..len).step_by(2) {
                let word = (reply[i] as u32) << 8 | (reply[i+1] as u32);
                sum = sum.wrapping_add(word);
            }
            while (sum >> 16) > 0 { sum = (sum & 0xFFFF) + (sum >> 16); }
            let checksum = !(sum as u16);
            reply[36] = (checksum >> 8) as u8;
            reply[37] = (checksum & 0xFF) as u8;
            transmit_raw_packet(&reply[0..len]);
        }
    } 
    else if protocol == 17 { 
        let ip_header_len = (packet[14] & 0x0F) as usize * 4;
        let udp_header_start = 14 + ip_header_len;
        
        if packet.len() >= udp_header_start + 8 {
            let dst_port = (packet[udp_header_start + 2] as u16) << 8 | (packet[udp_header_start + 3] as u16);

// 🎯 BARE-METAL WEBASSEMBLY (WASM) INTERPRETER

pub const OP_I32_CONST: u8 = 0x41; // Push a 32-bit integer to the stack
pub const OP_I32_ADD: u8   = 0x6A; // Pop two numbers, add them, push result
pub const OP_I32_MUL: u8   = 0x6C; // Pop two numbers, multiply them, push result
pub const OP_END: u8       = 0x0B; // End of function/execution

pub fn validate_and_execute(bytecode: &[u8]) -> Option<i32> {
    let mut stack: [i32; 64] = [0; 64]; // Secure Fixed-Size Stack
    let mut sp = 0; // Stack Pointer
    let mut pc = 0; // Program Counter

    while pc < bytecode.len() {
        let op = bytecode[pc];
        pc += 1;

        match op {
            OP_I32_CONST => {
                if pc >= bytecode.len() { return None; } // Sandbox Protection: Out of bounds
                stack[sp] = bytecode[pc] as i32;
                sp += 1;
                pc += 1;
            },
            OP_I32_ADD => {
                if sp < 2 { return None; } // Sandbox Protection: Stack Underflow
                let b = stack[sp - 1];
                let a = stack[sp - 2];
                stack[sp - 2] = a.wrapping_add(b); // Prevent overflow crashes
                sp -= 1;
            },
            OP_I32_MUL => {
                if sp < 2 { return None; }
                let b = stack[sp - 1];
                let a = stack[sp - 2];
                stack[sp - 2] = a.wrapping_mul(b);
                sp -= 1;
            },
            OP_END => {
                if sp > 0 { return Some(stack[sp - 1]); }
                return Some(0);
            },
            _ => return None, // Invalid Opcode: Kill the execution safely
        }
    }
    None // Execution finished without END opcode
}

### Opt-in for performance

[bufferutil][] is an optional module that can be installed alongside the ws
module:

```
npm install --save-optional bufferutil
```

This is a binary addon that improves the performance of certain operations such
as masking and unmasking the data payload of the WebSocket frames. Prebuilt
binaries are available for the most popular platforms, so you don't necessarily
need to have a C++ compiler installed on your machine.

To force ws to not use bufferutil, use the
[`WS_NO_BUFFER_UTIL`](./doc/ws.md#ws_no_buffer_util) environment variable. This
can be useful to enhance security in systems where a user can put a package in
the package search path of an application of another user, due to how the
Node.js resolver algorithm works.

#### Legacy opt-in for performance

If you are running on an old version of Node.js (prior to v18.14.0), ws also
supports the [utf-8-validate][] module:

```
npm install --save-optional utf-8-validate
```

This contains a binary polyfill for [`buffer.isUtf8()`][].

To force ws not to use utf-8-validate, use the
[`WS_NO_UTF_8_VALIDATE`](./doc/ws.md#ws_no_utf_8_validate) environment variable.

## WebSocket compression

ws supports the [permessage-deflate extension][permessage-deflate] which enables
the client and server to negotiate a compression algorithm and its parameters,
and then selectively apply it to the data payloads of each WebSocket message.

The extension is disabled by default on the server and enabled by default on the
client. It adds a significant overhead in terms of performance and memory
consumption so we suggest to enable it only if it is really needed.

Note that Node.js has a variety of issues with high-performance compression,
where increased concurrency, especially on Linux, can lead to [catastrophic
memory fragmentation][node-zlib-bug] and slow performance. If you intend to use
permessage-deflate in production, it is worthwhile to set up a test
representative of your workload and ensure Node.js/zlib will handle it with
acceptable performance and memory usage.

Tuning of permessage-deflate can be done via the options defined below. You can
also use `zlibDeflateOptions` and `zlibInflateOptions`, which is passed directly
into the creation of [raw deflate/inflate streams][node-zlib-deflaterawdocs].

See [the docs][ws-server-options] for more options.

```js
import WebSocket, { WebSocketServer } from 'ws';

const wss = new WebSocketServer({
  port: 8080,
  perMessageDeflate: {
    zlibDeflateOptions: {
      // See zlib defaults.
      chunkSize: 1024,
      memLevel: 7,
      level: 3
    },
    zlibInflateOptions: {
      chunkSize: 10 * 1024
    },
    // Other options settable:
    clientNoContextTakeover: true, // Defaults to negotiated value.
    serverNoContextTakeover: true, // Defaults to negotiated value.
    serverMaxWindowBits: 10, // Defaults to negotiated value.
    // Below options specified as default values.
    concurrencyLimit: 10, // Limits zlib concurrency for perf.
    threshold: 1024 // Size (in bytes) below which messages
    // should not be compressed if context takeover is disabled.
  }
});
```

The client will only use the extension if it is supported and enabled on the
server. To always disable the extension on the client, set the
`perMessageDeflate` option to `false`.

```js
import WebSocket from 'ws';

const ws = new WebSocket('ws://www.host.com/path', {
  perMessageDeflate: false
});
```

### Client authentication

    if (err || !client) {
      socket.write('HTTP/1.1 401 Unauthorized\r\n\r\n');
      socket.destroy();
      return;

    socket.removeListener('error', onSocketError);

    wss.handleUpgrade(request, socket, head, function done(ws) {
      wss.emit('connection', ws, request, client);
    });
  });

### Server broadcast

A client WebSocket broadcasting to all connected WebSocket clients, including

```js
import WebSocket, { WebSocketServer } from 'ws';

const wss = new WebSocketServer({ port: 8080 });

      }
    });
  });
});
```

A client WebSocket broadcasting to every other connected WebSocket clients,
excluding itself.

```js
import WebSocket, { WebSocketServer } from 'ws';

const wss = new WebSocketServer({ port: 8080 });

wss.on('connection', function connection(ws) {
  ws.on('error', console.error);

  ws.on('message', function message(data, isBinary) {
    wss.clients.forEach(function each(client) {
      if (client !== ws && client.readyState === WebSocket.OPEN) {
        client.send(data, { binary: isBinary });
      }
    });
  });
});
```

### Round-trip time

```js
import WebSocket from 'ws';

const ws = new WebSocket('wss://websocket-echo.com/');

ws.on('error', console.error);

ws.on('open', function open() {
  console.log('connected');
  ws.send(Date.now());
});

ws.on('close', function close() {
  console.log('disconnected');
});

ws.on('message', function message(data) {
  console.log(`Round-trip time: ${Date.now() - data} ms`);

  setTimeout(function timeout() {
    ws.send(Date.now());
  }, 500);
});
```

## FAQ

### How to get the IP address of the client?

The remote IP address can be obtained from the raw socket.

```js
import { WebSocketServer } from 'ws';

const wss = new WebSocketServer({ port: 8080 });

wss.on('connection', function connection(ws, req) {
  const ip = req.socket.remoteAddress;

  ws.on('error', console.error);
});
```

When the server runs behind a proxy like NGINX, the de-facto standard is to use
the `X-Forwarded-For` header.

```js
wss.on('connection', function connection(ws, req) {
  const ip = req.headers['x-forwarded-for'].split(',')[0].trim();

  ws.on('error', console.error);
});
```

### How to detect and close broken connections?

Sometimes, the link between the server and the client can be interrupted in a
way that keeps both the server and the client unaware of the broken state of the
connection (e.g. when pulling the cord).

In these cases, ping messages can be used as a means to verify that the remote
endpoint is still responsive.

```js
import { WebSocketServer } from 'ws';

function heartbeat() {
  this.isAlive = true;
}

const wss = new WebSocketServer({ port: 8080 });

wss.on('connection', function connection(ws) {
  ws.isAlive = true;
  ws.on('error', console.error);
  ws.on('pong', heartbeat);
});

const interval = setInterval(function ping() {
  wss.clients.forEach(function each(ws) {
    if (ws.isAlive === false) return ws.terminate();

    ws.isAlive = false;
    ws.ping();
  });
}, 30000);

wss.on('close', function close() {
  clearInterval(interval);
});
```

Pong messages are automatically sent in response to ping messages as required by
the spec.

Just like the server example above, your clients might as well lose connection
without knowing it. You might want to add a ping listener on your clients to
prevent that. A simple implementation would be:

```js
import WebSocket from 'ws';

function heartbeat() {
  clearTimeout(this.pingTimeout);

  // Use `WebSocket#terminate()`, which immediately destroys the connection,
  // instead of `WebSocket#close()`, which waits for the close timer.
  // Delay should be equal to the interval at which your server
  // sends out pings plus a conservative assumption of the latency.
  this.pingTimeout = setTimeout(() => {
    this.terminate();
  }, 30000 + 1000);
}

const client = new WebSocket('wss://websocket-echo.com/');

client.on('error', console.error);
client.on('open', heartbeat);
client.on('ping', heartbeat);
client.on('close', function clear() {
  clearTimeout(this.pingTimeout);
});
```
